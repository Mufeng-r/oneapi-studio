换个 base_url，就接完了

三类能力共用一套鉴权与错误码。已有的 OpenAI SDK 代码不用重写。

# 文生图 — 换 base_url 即可
from openai import OpenAI

client = OpenAI(
    base_url="https://api.oneapi.studio/v1",
    api_key="sk-oneapi-...",
)

img = client.images.generate(
    model="flux-1.1-pro",
    prompt="霓虹雨夜的赛博东京街头，电影感",
    size="1024x1024",
    n=4,
)
print(img.data[0].url)
# 文生视频 — 异步任务，轮询或 Webhook
import requests

task = requests.post(
    "https://api.oneapi.studio/v1/videos/generations",
    headers={"Authorization": "Bearer sk-oneapi-..."},
    json={
        "model":    "kling-v2-master",
        "prompt":   "航拍穿过云层，日出破晓",
        "duration": 5,
        "webhook":  "https://your.app/hook",
    },
).json()

print(task["task_id"], task["status"])  # queued
// Coding — 原生 OpenAI SDK，流式输出
import OpenAI from "openai";

const client = new OpenAI({
  baseURL: "https://api.oneapi.studio/v1",
  apiKey:  process.env.ONEAPI_KEY,
});

const stream = await client.chat.completions.create({
  model:    "claude-opus-5-5",   // 或 gpt-5 / deepseek-v3
  messages: [{ role: "user", content: "重构这个函数并加测试" }],
  stream:   true,
});

for await (const chunk of stream) {
  process.stdout.write(chunk.choices[0]?.delta?.content ?? "");
}
